! ================================================================
! Device: F3-SW (Floor 3 Switch - Cisco Catalyst 2960)
! Role: Floor 3 Access Switch (Sales, AP & Multicast Receiver)
! Features: VLAN 30, Port Security (Max 2, Shutdown), BPDU Guard,
!           DHCP Snooping, IGMP Snooping, Native VLAN 999 Trunk
! ================================================================
enable
configure terminal
hostname F3-SW

! 1. Create VLANs
vlan 30
 name Sales
vlan 99
 name Management
vlan 999
 name Native
exit

! 2. Trunk Uplink to HQ-Core-SW
interface GigabitEthernet0/1
 description Uplink to HQ-Core-SW
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,99,999
 no shutdown
 exit

! 3. Management SVI
interface vlan 99
 ip address 192.168.10.181 255.255.255.248
 no shutdown
 exit
ip default-gateway 192.168.10.177

! 4. Access Ports for Sales PCs, AP, and Multicast Receivers (Fa0/1 - Fa0/15)
interface range FastEthernet0/1 - 15
 description Sales-Users-and-AP
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
 no shutdown
 exit

! 5. DHCP Snooping
ip dhcp snooping
ip dhcp snooping vlan 30,99
no ip dhcp snooping information option

interface GigabitEthernet0/1
 ip dhcp snooping trust
 exit

! 6. IGMP Snooping for Multicast Video Stream
ip igmp snooping
ip igmp snooping vlan 30

! 7. Disable Unused Ports
interface range FastEthernet0/16 - 24, GigabitEthernet0/2
 shutdown
 exit

end
write memory
